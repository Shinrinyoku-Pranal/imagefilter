cmake_minimum_required(VERSION 3.20)
project(ImageFilter VERSION 1.0.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(imagefilter_core src/image.cpp src/filters.cpp src/thread_pool.cpp src/codecs.cpp)
target_include_directories(imagefilter_core PUBLIC include)
target_include_directories(imagefilter_core SYSTEM PRIVATE third_party)
target_compile_features(imagefilter_core PUBLIC cxx_std_17)
target_link_libraries(imagefilter_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(imagefilter_core PRIVATE /W4)
else()
  target_compile_options(imagefilter_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(imagefilter src/main.cpp)
target_link_libraries(imagefilter PRIVATE imagefilter_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(imagefilter_tests tests/tests.cpp)
  target_link_libraries(imagefilter_tests PRIVATE imagefilter_core)
  add_test(NAME unit COMMAND imagefilter_tests)
  find_package(Python3 3.9 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:imagefilter>)
    add_test(NAME scripts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/scripts_test.py $<TARGET_FILE:imagefilter>)
  endif()
endif()
